create extension if not exists pgcrypto; create table public.profiles ( id uuid primary key references auth.users(id) on delete cascade, full_name text not null, phone text, is_admin boolean default false, created_at timestamptz default now() ); create table public.surveys ( id uuid primary key default gen_random_uuid(), title text not null, description text, category text default 'Health', access_fee numeric(10,2) not null default 100, published boolean default false, created_at timestamptz default now() ); create table public.questions ( id uuid primary key default gen_random_uuid(), survey_id uuid references public.surveys(id) on delete cascade, question_text text not null, question_type text default 'text', options jsonb, created_at timestamptz default now() ); create table public.payments ( id uuid primary key default gen_random_uuid(), user_id uuid references auth.users(id), survey_id uuid references public.surveys(id), amount numeric(10,2) not null default 100, phone text not null, mpesa_receipt text, status text not null default 'pending', created_at timestamptz default now(), constraint valid_payment_status check (status in ('pending','confirmed','rejected')) ); create table public.responses ( id uuid primary key default gen_random_uuid(), user_id uuid references auth.users(id), survey_id uuid references public.surveys(id), question_id uuid references public.questions(id), answer text, created_at timestamptz default now() ); alter table public.responses enable row level security; create or replace function public.is_admin() returns boolean language sql security definer set search_path = public stable as $$ select exists ( select 1 from public.profiles where id = auth.uid() and is_admin = true ); $$; create policy "Read own profile" on public.profiles for select using (id = auth.uid() or public.is_admin()); create policy "Create own profile" on public.profiles for insert with check (id = auth.uid()); create policy "Admin manages profiles" on public.profiles for update using (public.is_admin()); create policy "Read published surveys" on public.surveys for select using (published = true or public.is_admin()); create policy "Admin manages surveys" on public.surveys for all using (public.is_admin()) with check (public.is_admin()); create policy "Paid users read questions" on public.questions for select using ( public.is_admin() or exists ( select 1 from public.payments p where p.user_id = auth.uid() and p.survey_id = questions.survey_id and p.status = 'confirmed' ) ); create policy "Users submit pending payments" on public.payments for insert with check ( user_id = auth.uid() and status = 'pending' and amount = 100 ); create policy "Users read own payments" on public.payments for select using (user_id = auth.uid() or public.is_admin()); create policy "Admin verifies payments" on public.payments for update using (public.is_admin()) with check (public.is_admin()); create policy "Users read own responses" on public.responses for select using (user_id = auth.uid() or public.is_admin()); create policy "Paid users submit responses" on public.responses for insert with check ( user_id = auth.uid() and exists ( select 1 from public.payments p where p.user_id = auth.uid() and p.survey_id = responses.survey_id and p.status = 'confirmed' ) );
